#include <tunables/global>

profile pihole flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Deliberately broad on files: Pi-hole's FTL, gravity and the bundled
  # cron/logrotate jobs touch a wide part of the filesystem. The value of this
  # profile is capability and mount confinement, not path confinement.
  file,
  signal,
  ptrace,

  # No new privileges, no mounting, no kernel module or raw device access.
  deny @{PROC}/sys/kernel/** w,
  deny mount,
  deny umount,
  deny /sys/kernel/security/** rwklx,
  deny /sys/firmware/** rwklx,

  # Supervisor grants no extra capabilities to this App, so the container only
  # ever holds Docker's default set. Enumerating that set here just makes the
  # profile break when the base image needs one more of them, so allow the
  # default set and deny the ones that would actually widen the App's reach.
  # Pi-hole needs setfcap to put file capabilities on pihole-FTL, which is how
  # FTL binds port 53 while running as a non-root user.
  capability,

  # DHCP is out of scope for this App. FTL cannot configure interfaces.
  deny capability net_admin,

  # FTL needs raw sockets and netlink for DNS, ARP and interface discovery.
  network,
}
